##三、代码规范
 
在网络上学习了有关编写代码的基础课程，掌握了基础的一些编程语言和代码编写方法。为保证代码的可读性、可维护性与团队协作效率，电控组遵循以下代码规范（基于STM32CubeIDE开发环境）。
 
1. 命名规则
 
- 变量/函数名：采用 小驼峰式 ，如 motorSpeed 、 servoSetAngle 。
- 宏定义/常量：全大写+下划线，如 MAX_PWM_VALUE 、 SERVO_PIN 。
- 模块前缀：不同硬件模块添加前缀区分，如 motor_ （电机）、 servo_ （舵机）、 ps2_ （遥控器）。
 
2. 代码格式
 
- 缩进：使用4个空格（禁止Tab混合空格）。
- 括号：左括号不换行，右括号与 else / while 等关键字同行，示例：
c
  
if (condition) {
    // 代码块
} else {
    //  else分支
}
 
- 空行分隔：不同功能逻辑块之间用空行分隔，增强可读性。
 
3. 注释要求
 
- 函数注释：使用多行注释 /* ... */ ，说明功能、参数、返回值，示例：
c
  
/*
 * @brief  麦克纳姆轮全向移动控制
 * @param  vx: X方向速度（范围：-10~10）
 * @param  vy: Y方向速度（范围：-10~10）
 * @param  omega: 旋转角速度（范围：-5~5）
 * @retval None
 */
void OmniMove(float vx, float vy, float omega);
 
- 关键逻辑注释：单行注释 // 说明核心算法、硬件映射等，示例：
c
  
// 运动学解算：根据麦克纳姆轮布局（前左、前右、后左、后右）
wheel_speed[0] = vx - vy - omega;
wheel_speed[1] = vx + vy + omega;
 
 
4. 工程结构
 
- 文件夹划分：按模块创建子文件夹，如 /Motor/ 、 /Servo/ 、 /PS2/ ，每个文件夹包含 .c 和 .h 文件。
- 头文件保护：使用宏定义防止头文件重复包含，示例：
c
  
// Motor.h
#ifndef __MOTOR_H
#define __MOTOR_H
// 内容...
#endif
 
 
二、电机驱动模块
 
功能描述
 
实现麦克纳姆轮的差速运动控制，支持机器人全向移动（前进、后退、横移、旋转）。
 
开发历程
 
1. 硬件接线测试
基于官方物资中要求的 L298N 驱动板与 TT马达 ，完成接线：
- L298N使能端接STM32的 PWM输出引脚 ，控制电机转速；
- 方向引脚接 PA0 、 PA1 ，控制电机正反转；
- 电源端接入12V电池，电机输出端接麦克纳姆轮的TT马达。
2. 基础驱动代码编写
c
  
// Motor.c
#include "Motor.h"

/* 宏定义：PWM周期（对应STM32定时器ARR值） */
#define PWM_PERIOD 1000

void Motor_Init(void) {
    /* 初始化PWM与GPIO */
    HAL_TIM_PWM_Start(&htim1, TIM_CHANNEL_1); // 使能PWM输出
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_0, GPIO_PIN_SET);  // 初始方向
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_1, GPIO_PIN_RESET);
}

void Motor_SetSpeed(uint8_t motor_id, uint8_t speed) {
    /* 
     * @brief  单个电机调速
     * @param  motor_id: 电机编号（0-3，对应麦克纳姆轮四马达）
     * @param  speed: 速度值（0-100）
     */
    uint16_t pwm_val = (speed * PWM_PERIOD) / 100; // 占空比映射
    switch (motor_id) {
        case 0:
            __HAL_TIM_SET_COMPARE(&htim1, TIM_CHANNEL_1, pwm_val);
            break;
        // 其他电机通道...
        default:
            break;
    }
}
 
3. 麦克纳姆轮运动学解算
推导全向移动的速度分解公式，编写四电机差速控制函数：
c
  
void OmniMove(float vx, float vy, float omega) {
    /* 
     * @brief  麦克纳姆轮全向移动控制
     * @param  vx: X方向速度（范围：-10~10）
     * @param  vy: Y方向速度（范围：-10~10）
     * @param  omega: 旋转角速度（范围：-5~5）
     */
    float wheel_speed[4];
    // 运动学解算（麦克纳姆轮布局：前左、前右、后左、后右）
    wheel_speed[0] = vx - vy - omega;
    wheel_speed[1] = vx + vy + omega;
    wheel_speed[2] = vx + vy - omega;
    wheel_speed[3] = vx - vy + omega;

    // 归一化与限幅
    float max_speed = 0;
    for (int i = 0; i < 4; i++) {
        if (fabs(wheel_speed[i]) > max_speed) {
            max_speed = fabs(wheel_speed[i]);
        }
    }
    if (max_speed > 10.0f) { // 超过最大速度则等比例缩小
        for (int i = 0; i < 4; i++) {
            wheel_speed[i] /= max_speed / 10.0f;
        }
    }

    // 输出到四个电机
    for (int i = 0; i < 4; i++) {
        Motor_SetSpeed(i, (uint8_t)(fabs(wheel_speed[i]) * 10));
        // 控制方向（根据速度正负）
        if (wheel_speed[i] < 0) {
            HAL_GPIO_WritePin(GPIOA, MOTOR_DIR_PINS[i*2], GPIO_PIN_RESET);
            HAL_GPIO_WritePin(GPIOA, MOTOR_DIR_PINS[i*2+1], GPIO_PIN_SET);
        } else {
            HAL_GPIO_WritePin(GPIOA, MOTOR_DIR_PINS[i*2], GPIO_PIN_SET);
            HAL_GPIO_WritePin(GPIOA, MOTOR_DIR_PINS[i*2+1], GPIO_PIN_RESET);
        }
    }
}
 
4. 调试问题与解决
- 问题：电机启动时抖动严重。
解决：在 Motor_Init 中增加PWM死区时间配置，并优化电源回路的滤波电容。
- 问题：横移时两侧电机不同步。
解决：在 OmniMove 中加入PID调速（单独模块实现），校准每个电机的转速系数。
 
三、舵机控制模块
 
功能描述
 
控制 MG995 和 SG90 舵机，实现机械臂或机构的角度精准调节。
 
开发历程
 
1. PWM信号生成
SG90舵机需要50Hz PWM（占空比对应0°~180°），使用STM32的定时器生成：
c
  
// Servo.c
#include "Servo.h"

/* 宏定义：舵机PWM参数 */
#define SERVO_PWM_PERIOD 20000   // 20ms周期（50Hz）
#define SERVO_MIN_DUTY  1000    // 0°对应占空比（1ms）
#define SERVO_MAX_DUTY  2000    // 180°对应占空比（2ms）

void Servo_Init(void) {
    HAL_TIM_PWM_Start(&htim2, TIM_CHANNEL_2); // 配置TIM2输出50Hz PWM
}

void Servo_SetAngle(uint8_t servo_id, uint8_t angle) {
    /* 
     * @brief  舵机角度设置
     * @param  servo_id: 舵机编号（0:MG995；1:SG90）
     * @param  angle: 目标角度（0~180）
     */
    uint16_t duty = SERVO_MIN_DUTY + (angle * (SERVO_MAX_DUTY - SERVO_MIN_DUTY)) / 180;
    switch (servo_id) {
        case 0:
            __HAL_TIM_SET_COMPARE(&htim2, TIM_CHANNEL_2, duty);
            break;
        case 1:
            __HAL_TIM_SET_COMPARE(&htim2, TIM_CHANNEL_3, duty);
            break;
        default:
            break;
    }
}
四、遥控器通信模块
功能描述
 
通过 PS2接收器 接收手柄指令，实现机器人的手动控制。
 
开发历程
 
1. SPI通信初始化
PS2接收器采用SPI协议，配置STM32的SPI接口：
// PS2.c
#include "PS2.h"

void PS2_Init(void) {
    HAL_SPI_Init(&hspi1);
    // 配置片选引脚为输出（高电平默认不选中）
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_4, GPIO_PIN_SET);
}

uint8_t PS2_ReadData(void) {
    /* @brief 读取PS2接收器的1字节数据 */
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_4, GPIO_PIN_RESET); // 拉低片选，选中设备
    uint8_t data;
    HAL_SPI_Receive(&hspi1, &data, 1, 100); // 阻塞式读取1字节
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_4, GPIO_PIN_SET); // 释放片选
    return data;
}
2.手柄指令解析
对于PS2手柄的按键遥杆数据映射到机器人动作：
typedef struct {
    uint8_t up;    // 上方向键
    uint8_t down;  // 下方向键
    uint8_t left;  // 左方向键
    uint8_t right; // 右方向键
    uint8_t l2;    // L2按键（调速档位）
} PS2_Command;

PS2_Command PS2_Parse(void) {
    /* @brief 解析PS2手柄原始数据，返回指令结构体 */
    PS2_Command cmd = {0};
    uint8_t raw_data = PS2_ReadData();
    // 解析原始数据（根据PS2通信协议映射按键）
    cmd.up = (raw_data & 0x01) ? 1 : 0;
    cmd.down = (raw_data & 0x02) ? 1 : 0;
    cmd.left = (raw_data & 0x04) ? 1 : 0;
    cmd.right = (raw_data & 0x08) ? 1 : 0;
    cmd.l2 = (raw_data & 0x10) ? 1 : 0;
    return cmd;
}

void Remote_Control(void) {
    /* @brief 遥控器控制主逻辑 */
    PS2_Command cmd = PS2_Parse();
    if (cmd.up) {
        OmniMove(10, 0, 0); // 前进（X方向速度10）
    } else if (cmd.down) {
        OmniMove(-10, 0, 0); // 后退（X方向速度-10）
    } else if (cmd.left) {
        OmniMove(0, -10, 0); // 左移（Y方向速度-10）
    } else if (cmd.right) {
        OmniMove(0, 10, 0);  // 右移（Y方向速度10）
    }
    // L2按键切换速度档位
    if (cmd.l2) {
        global_speed_scale = 1.5f; // 速度放大1.5倍
    } else {
        global_speed_scale = 1.0f;
    }
}
